class Solution {
public:

    bool connect(int x,int y, int a, int b,
                 vector<vector<int>>& arr,
                 int m ,int n,
                 vector<vector<int>>& visited){

        visited[x][y]=1;

        if(x==a&&y==b){
            return true;
        }

        // up
        if((x-1)>=0&&arr[x-1][y]==1&&visited[x-1][y]==0){
            if(connect(x-1,y,a,b,arr,m,n,visited)){
                return true;
            }else{
                visited[x-1][y]=0;
            }
        }

        // down
        if((x+1)<m&&arr[x+1][y]==1&&visited[x+1][y]==0){
            if(connect(x+1,y,a,b,arr,m,n,visited)){
                return true;
            }else{
                visited[x+1][y]=0;
            }
        }

        // left
        if((y-1)>=0&&arr[x][y-1]==1&&visited[x][y-1]==0){
            if(connect(x,y-1,a,b,arr,m,n,visited)){
                return true;
            }else{
                visited[x][y-1]=0;
            }
        }

        // right
        if((y+1)<n&&arr[x][y+1]==1&&visited[x][y+1]==0){
            if(connect(x,y+1,a,b,arr,m,n,visited)){
                return true;
            }else{
                visited[x][y+1]=0;
            }
        }

        return false;
    }






    void check(int x, int y,
               vector<vector<int>>& arr,
               vector<int>& p,
               int n,int m){

        vector<vector<int>> visited(m,vector<int>(n,0));

        vector<int> count(4,0);

        if(arr[x][y]==1){

            if(p.size()==0){
                p.push_back(1);
            }else{
                p.push_back(p[p.size()-1]);
            }

            return;
        }

        else{

            // down
            if((x+1)<m&&(arr[x+1][y]==1)){
                count[0]++;
            }

            // right
            if((y+1)<n&&(arr[x][y+1]==1)){
                count[1]++;
            }

            // up
            if((x-1)>=0&&(arr[x-1][y]==1)){
                count[2]++;
            }

            // left
            if((y-1)>=0&&(arr[x][y-1]==1)){
                count[3]++;
            }

            int sum=count[0]+count[1]+count[2]+count[3];



            if(sum==0){

                arr[x][y]=1;

                if(p.size()==0){
                    p.push_back(1);
                }else{
                    p.push_back(p[p.size()-1]+1);
                }
            }



            else if(sum==1){

                p.push_back(p[p.size()-1]);

                arr[x][y]=1;
            }



            else if(sum==2){

                if((count[0]+count[1])==2){

                    if(connect(x+1,y,x,y+1,arr,m,n,visited)){
                        p.push_back(p[p.size()-1]);
                    }else{
                        p.push_back(p[p.size()-1]-1);
                    }

                    arr[x][y]=1;
                }

                else if((count[1]+count[2])==2){

                    if(connect(x,y+1,x-1,y,arr,m,n,visited)){
                        p.push_back(p[p.size()-1]);
                    }else{
                        p.push_back(p[p.size()-1]-1);
                    }

                    arr[x][y]=1;
                }

                else if((count[2]+count[3])==2){

                    if(connect(x-1,y,x,y-1,arr,m,n,visited)){
                        p.push_back(p[p.size()-1]);
                    }else{
                        p.push_back(p[p.size()-1]-1);
                    }

                    arr[x][y]=1;
                }

                else if((count[0]+count[3])==2){

                    if(connect(x+1,y,x,y-1,arr,m,n,visited)){
                        p.push_back(p[p.size()-1]);
                    }else{
                        p.push_back(p[p.size()-1]-1);
                    }

                    arr[x][y]=1;
                }

                else if((count[1]+count[3])==2){

                    if(connect(x,y+1,x,y-1,arr,m,n,visited)){
                        p.push_back(p[p.size()-1]);
                    }else{
                        p.push_back(p[p.size()-1]-1);
                    }

                    arr[x][y]=1;
                }

                else if((count[0]+count[2])==2){

                    if(connect(x+1,y,x-1,y,arr,m,n,visited)){
                        p.push_back(p[p.size()-1]);
                    }else{
                        p.push_back(p[p.size()-1]-1);
                    }

                    arr[x][y]=1;
                }
            }




            else if(sum==3){

                int connected=0;

                // 0 1
                if(count[0]&&count[1]){
                    vector<vector<int>> v1(m,vector<int>(n,0));

                    if(connect(x+1,y,x,y+1,arr,m,n,v1)){
                        connected++;
                    }
                }

                // 0 2
                if(count[0]&&count[2]){
                    vector<vector<int>> v2(m,vector<int>(n,0));

                    if(connect(x+1,y,x-1,y,arr,m,n,v2)){
                        connected++;
                    }
                }

                // 0 3
                if(count[0]&&count[3]){
                    vector<vector<int>> v3(m,vector<int>(n,0));

                    if(connect(x+1,y,x,y-1,arr,m,n,v3)){
                        connected++;
                    }
                }

                // 1 2
                if(count[1]&&count[2]){
                    vector<vector<int>> v4(m,vector<int>(n,0));

                    if(connect(x,y+1,x-1,y,arr,m,n,v4)){
                        connected++;
                    }
                }

                // 1 3
                if(count[1]&&count[3]){
                    vector<vector<int>> v5(m,vector<int>(n,0));

                    if(connect(x,y+1,x,y-1,arr,m,n,v5)){
                        connected++;
                    }
                }

                // 2 3
                if(count[2]&&count[3]){
                    vector<vector<int>> v6(m,vector<int>(n,0));

                    if(connect(x-1,y,x,y-1,arr,m,n,v6)){
                        connected++;
                    }
                }

                if(connected==0){
                    p.push_back(p[p.size()-1]-2);
                }

                else if(connected==1){
                    p.push_back(p[p.size()-1]-1);
                }

                else{
                    p.push_back(p[p.size()-1]);
                }

                arr[x][y]=1;
            }






            else if(sum==4){

                int connected=0;

                {
                    vector<vector<int>> v1(m,vector<int>(n,0));

                    if(connect(x+1,y,x,y+1,arr,m,n,v1)){
                        connected++;
                    }
                }

                {
                    vector<vector<int>> v2(m,vector<int>(n,0));

                    if(connect(x+1,y,x-1,y,arr,m,n,v2)){
                        connected++;
                    }
                }

                {
                    vector<vector<int>> v3(m,vector<int>(n,0));

                    if(connect(x+1,y,x,y-1,arr,m,n,v3)){
                        connected++;
                    }
                }

                {
                    vector<vector<int>> v4(m,vector<int>(n,0));

                    if(connect(x,y+1,x-1,y,arr,m,n,v4)){
                        connected++;
                    }
                }

                {
                    vector<vector<int>> v5(m,vector<int>(n,0));

                    if(connect(x,y+1,x,y-1,arr,m,n,v5)){
                        connected++;
                    }
                }

                {
                    vector<vector<int>> v6(m,vector<int>(n,0));

                    if(connect(x-1,y,x,y-1,arr,m,n,v6)){
                        connected++;
                    }
                }

                if(connected==0){
                    p.push_back(p[p.size()-1]-3);
                }

                else if(connected<=2){
                    p.push_back(p[p.size()-1]-2);
                }

                else if(connected<=4){
                    p.push_back(p[p.size()-1]-1);
                }

                else{
                    p.push_back(p[p.size()-1]);
                }

                arr[x][y]=1;
            }
        }
    }






    vector<int> numOfIslands(int n, int m,
                             vector<vector<int>>& operators) {

        vector<vector<int>> arr(m,vector<int>(n,0));

        vector<int> count;

        int x;
        int y;

        for(int i=0;i<operators.size();i++){

            x=operators[i][0];
            y=operators[i][1];

            check(x,y,arr,count,n,m);
        }

        return count;
    }
