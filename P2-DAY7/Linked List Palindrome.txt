Node* reverse(Node* head){
    Node* temp=head;
    Node* temp2=head->next;
    head->next=NULL;
    
    while(temp2!=NULL){
        temp=temp2;
        temp2=temp2->next;
        temp->next=head;
        head=temp;
    }
    return head;
}
Node* midl(Node* head){
    if(head->next==NULL){
        return head;
    }
    Node* slow=head;
    Node* fast=head->next;
    int count=0;
    while(fast->next!=NULL&&fast->next->next!=NULL){
        fast=fast->next->next;
        slow=slow->next;
        count++;
    }
    return slow;
}
int midl2(Node* head){

    Node* slow=head;

    int count=0;
    while(fast->next!=NULL&&fast->next->next!=NULL){

        slow=slow->next;
        count++;
    }
    if(fast->next==NULL){

}
